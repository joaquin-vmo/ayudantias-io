\begin{center}
\begin{tikzpicture}[scale=\bpsc,x=0.4cm,y=0.4cm,>=stealth,font=\small,line cap=round]
  \def\e{\bpe}
  \pgfmathsetmacro{\xmax}{#4+3}
  \draw[->] (0,0) -- (\xmax,0) node[right] {$P_1$};
  \draw[->] (0,0) -- (0,\xmax) node[above] {$P_2$};
  \draw[densely dotted,gray] (0,0) -- (\xmax-0.5,\xmax-0.5);
  \draw[densely dashed,gray!70] (#3,0) -- (#3,#3) -- (0,#3);
  \draw[densely dashed,gray!70] (#4,0) -- (#4,#4) -- (0,#4);
  \draw (#1,0.15) -- (#1,-0.15) node[below] {#5};
  \draw (#3,0.15) -- (#3,-0.15) node[below] {#7};
  \draw (0.15,#2) -- (-0.15,#2) node[left] {#6};
  \draw (0.15,#4) -- (-0.15,#4) node[left] {#8};
  % Firma 1: P1(P2)
  \draw[thick,violet] (#1,0) -- (#1,#1+\e) -- (#3-\e,#3) -- (#3,#3) -- (#3,\xmax-0.5);
  % Firma 2: P2(P1)
  \draw[thick,blue!70] (0,#2) -- (#2+\e,#2) -- (#4,#4-\e) -- (#4,#4) -- (\xmax-0.5,#4);
  \fill[red] (#2-#9,#2) circle (2.5pt) node[below right,red] {$N$};
  \node[violet] at (#3-4,#3+1.6) {$P_1(P_2)$};
  \node[blue!70] at (#4+1.8,#4-5) {$P_2(P_1)$};
\end{tikzpicture}
\end{center}}
